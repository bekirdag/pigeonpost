Verify webhook signatures

Pigeonpost signs every webhook delivery so you can tell it came from your org's webhook and was not altered or replayed. Verify before you act on the body.

What arrives#

POST /hooks/pigeonpost HTTP/1.1
Content-Type: application/json
User-Agent: Pigeonpost-Webhooks/1
Pigeonpost-Event-Id: evt_…
Pigeonpost-Event-Type: member.joined
Pigeonpost-Delivery-Id: whd_…
Pigeonpost-Signature: t=1793095200,v1=5257a869e7ecebeda32affa62cdca3fa51cad7e77a0e56ff536d0ce8e108d8bd

{"id":"evt_…","type":"member.joined","created_at":"2026-10-27T10:15:00Z","org_id":"org_…","data":{"member_id":"mem_…","handle":"/acme/jane"}}

v1 is the hex HMAC-SHA256, keyed with your webhook's signing secret (the whole whsec_… string), over the bytes of t, a literal ., and the raw request body exactly as received. During the 24 hours after you rotate the secret, the header carries two v1= entries, the new secret's first; accept the delivery if either matches a secret you hold.

The rules#

  1. 1. Read the raw body bytes before any JSON parsing. Re-serialized JSON will not match.
  2. 2. Parse t and every v1 from the header.
  3. 3. Reject the delivery if t is more than 5 minutes away from your clock.
  4. 4. Compute the HMAC and compare in constant time against each v1.
  5. 5. Deduplicate on Pigeonpost-Event-Id (the body's id): delivery is at least once, and retries reuse the event id.
  6. 6. Answer 2xx within 10 seconds, then do the work. Anything else, including redirects, is retried with backoff for 24 hours.

Node.js#

import crypto from "node:crypto";

const TOLERANCE_SECONDS = 5 * 60;

// rawBody: the request body as a Buffer, exactly as received.
export function verifyPigeonpost(rawBody, header, secret, nowSeconds = Date.now() / 1000) {
  let t = null;
  const signatures = [];
  for (const part of String(header || "").split(",")) {
    const eq = part.indexOf("=");
    if (eq < 0) continue;
    const key = part.slice(0, eq).trim();
    const value = part.slice(eq + 1).trim();
    if (key === "t" && /^[0-9]+$/.test(value)) t = value;
    if (key === "v1" && /^[0-9a-f]{64}$/.test(value)) signatures.push(Buffer.from(value, "hex"));
  }
  if (t === null || signatures.length === 0) return false;
  if (Math.abs(nowSeconds - Number(t)) > TOLERANCE_SECONDS) return false;
  const expected = crypto.createHmac("sha256", secret).update(`${t}.`).update(rawBody).digest();
  return signatures.some((sig) => crypto.timingSafeEqual(sig, expected));
}

Python#

import hashlib
import hmac
import time

TOLERANCE_SECONDS = 5 * 60


def verify_pigeonpost(raw_body: bytes, header: str, secret: str, now: float | None = None) -> bool:
    """raw_body: the request body bytes exactly as received."""
    t, signatures = None, []
    for part in (header or "").split(","):
        key, sep, value = part.strip().partition("=")
        if not sep:
            continue
        if key == "t" and value.isascii() and value.isdigit():
            t = value
        elif key == "v1":
            signatures.append(value)
    if t is None or not signatures:
        return False
    if abs((time.time() if now is None else now) - int(t)) > TOLERANCE_SECONDS:
        return False
    expected = hmac.new(secret.encode(), t.encode() + b"." + raw_body, hashlib.sha256).hexdigest()
    return any(hmac.compare_digest(expected, sig) for sig in signatures)

Go#

package webhook

import (
	"crypto/hmac"
	"crypto/sha256"
	"encoding/hex"
	"strconv"
	"strings"
	"time"
)

const tolerance = 5 * time.Minute

// VerifyPigeonpost checks a Pigeonpost-Signature header against the raw request body.
func VerifyPigeonpost(rawBody []byte, header, secret string, now time.Time) bool {
	var ts string
	var signatures [][]byte
	for _, part := range strings.Split(header, ",") {
		key, value, ok := strings.Cut(strings.TrimSpace(part), "=")
		if !ok {
			continue
		}
		switch key {
		case "t":
			ts = value
		case "v1":
			if sig, err := hex.DecodeString(value); err == nil {
				signatures = append(signatures, sig)
			}
		}
	}
	seconds, err := strconv.ParseInt(ts, 10, 64)
	if err != nil || len(signatures) == 0 {
		return false
	}
	if age := now.Sub(time.Unix(seconds, 0)); age > tolerance || age < -tolerance {
		return false
	}
	mac := hmac.New(sha256.New, []byte(secret))
	mac.Write([]byte(ts + "."))
	mac.Write(rawBody)
	expected := mac.Sum(nil)
	for _, sig := range signatures {
		if hmac.Equal(sig, expected) {
			return true
		}
	}
	return false
}

Managing webhooks#

Create one with POST /webhooks (webhooks:write) or in the portal; the secret is shown once. Rotate with POST /webhooks/{webhook}/rotate, deploy the new secret, and the old one stops signing 24 hours later. GET /webhooks/{webhook}/deliveries shows recent attempts with their status (delivered, retrying, failed or blocked) and your endpoint's response code.

The URL must be https:// on port 443 or 8443 and resolve only to public addresses; it is checked at registration and at every delivery. The reference lists every event type and its fields: ids and handles only, never content.
